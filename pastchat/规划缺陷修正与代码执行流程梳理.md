### 3.2 dispatch 方法（行 81-113）

```typescript
async dispatch(
  name: AgentHookName,
  context: AgentHookContext,
): Promise<AgentHookResult<Record<string, unknown>>> {
  const queue = hooks.get(name) ?? [];
  let patch: Record<string, unknown> | undefined;
  const emittedEvents: Array<Record<string, unknown>> = [];

  for (const hook of queue) {
    const result = await hook(context);
    if (!result) continue;
    patch = mergeHookPatches(patch, result.patch);
    if (result.emittedEvents?.length) {
      emittedEvents.push(...result.emittedEvents);
    }
    if (result.blocked) {
      return { blocked: true, reason: result.reason, patch, emittedEvents };
    }
  }
  return { patch, emittedEvents };
}
```

```typescript
register('report_progress', async (context) => {
  const payload = (context.payload ?? {}) as {
    phase: ProgressPhase;
    agentType: string;
    taskDescription: string;
    taskIndex: number;
    parentTask: string | null;
    detail?: ProgressDetail;
  };

  const snapshot = context.snapshot as { sessionId?: string } | undefined;
  const saveId = snapshot?.sessionId;
  const requestId = context.requestId as string | undefined;

  // 排除无有效 saveId 的场景：未初始化的 Agent / 模板编辑器（saveId='0'）
  if (!saveId || saveId === 'unknown-session' || saveId === '0') return undefined;

  try {
    webSocketService.broadcastToSave(saveId, 'agent_progress', {
      phase: payload.phase,
      agentType: payload.agentType,
      taskDescription: payload.taskDescription,
      taskIndex: payload.taskIndex,
      parentTask: payload.parentTask,
      detail: payload.detail,
      timestamp: Date.now(),
    }, requestId);
  } catch (wsError) {
    logger.warn('report_progress WS broadcast failed', { error: wsError, saveId, phase: payload.phase });
  }

  return undefined;
});
```

```typescript
private startHeartbeat(): void {
  const heartbeatMs = appConfig.timeout.wsHeartbeat;
  const maxMissed = appConfig.timeout.wsMaxMissedHeartbeats;

  this.heartbeatInterval = setInterval(() => {
    for (const [ws, clientInfo] of this.clients) {
      if (!clientInfo.isAlive) {
        clientInfo.missedHeartbeats++;
        if (clientInfo.missedHeartbeats >= maxMissed) {
          logger.warn('Terminating dead WebSocket connection', {...});
          ws.terminate();
          this.clients.delete(ws);
          continue;
        }
      }
      clientInfo.isAlive = false;
      ws.ping();
    }
  }, heartbeatMs);
}
```

### 6.5 broadcastToSave 完整实现（行 238-265）

```typescript
broadcastToSave<T>(saveId: string, eventType: GameEventType, payload: T, requestId?: string): void {
  const message = {
    type: 'game:event' as const,
    requestId: requestId || '',
    module: 'game',
    eventType,
    data: { ...(payload as object) },
    intentHint: eventType,
    timestamp: Date.now(),
  };
  const data = JSON.stringify(message);
  let sent = 0;

  for (const [ws, clientInfo] of this.clients) {
    if (ws.readyState === WebSocket.OPEN && clientInfo.saveId === saveId) {
      try {
        ws.send(data);
        sent++;
      } catch (sendError) {
        logger.warn('WebSocket send failed for client', { error: ... });
      }
    }
  }

  if (sent > 0) {
    logger.debug(`Broadcast game:event(${eventType}) to save ${saveId}, ${sent} clients`);
  }
}
```

### 10.3 flush 方法完整逻辑（行 182-240）

```typescript
async flush(writeQueue: DatabaseWriteQueue, db: Knex): Promise<void> {
  if (this.writes.length === 0) return;

  logger.info('Flushing staged writes to DB', { count: this.writes.length });

  const failedWrites: StagedWrite[] = [];
  let successCount = 0;

  for (const op of this.writes) {
    try {
      await writeQueue.enqueueFn(
        () => this.executeWriteOp(db, op),
        `staging.${op.toolType}.${op.method}`,
      );
      successCount++;
    } catch (writeError) {
      logger.error('Staged write FAILED - program error detected', {...});
      failedWrites.push(op);
    }
  }

  logger.info('Staged writes flush completed', {...});

  // dev trace + broadcast
  const devTraceCollector = getDevTraceCollector();
  if (devTraceCollector && this._saveId) {
    devTraceCollector.addTrace(this._saveId, {...});
    try {
      webSocketService.broadcastToSave(this._saveId, 'dev:staging_commit', {...}, this._requestId);
    } catch (wsError) {...}
  }

  this.clear();  // ← 无论成功失败都 clear

  // 程序错误：flush 失败代表审核流程失效，必须报错停止
  if (failedWrites.length > 0) {
    throw new Error(`StagingPool flush: ${failedWrites.length}/${successCount + failedWrites.length} writes failed...`);
  }
}
```
